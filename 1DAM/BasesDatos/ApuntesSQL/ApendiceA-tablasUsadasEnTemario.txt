-- ============================================================================
-- THE EXPANDED BOOKS DATASET
-- ============================================================================
-- Main dataset used throughout Part 1. Includes 200 books by 76 authors.
-- Demonstrates proper database design with primary keys, foreign keys,
-- and many-to-many relationships via cross-reference tables.

-- View all tables in the dataset
SELECT * FROM authors;
SELECT * FROM books;
SELECT * FROM editions;  -- Only includes 4 books by J.R.R. Tolkien
SELECT * FROM awards;
SELECT * FROM authors_awards;  -- Cross-reference: authors to awards
SELECT * FROM books_awards;    -- Cross-reference: books to awards

-- Key relationships:
-- - Each book has one author (books.author_id → authors.author_id)
-- - Each book may have many editions (editions.book_id → books.book_id)
-- - Authors and awards have many-to-many relationship via authors_awards
-- - Books and awards have many-to-many relationship via books_awards

-- ============================================================================
-- THE BOOKSTORE DATASET
-- ============================================================================
-- Simulates a used book seller's database. Designed for data modification
-- practice. No foreign key constraints. One-to-one relationship between tables.

-- View all data
SELECT * FROM bookstore_inventory;
SELECT * FROM bookstore_sales;

-- Key points:
-- - stock_number is unique identifier in bookstore_inventory
-- - When a book sells, entry added to bookstore_sales
-- - Each sale record matches exactly one inventory record
-- - Books in sales table have been sold; others are still available

-- Find unsold books
SELECT * FROM bookstore_inventory
WHERE stock_number NOT IN (SELECT stock_number FROM bookstore_sales);

-- ============================================================================
-- NOTES ON DATABASE DIFFERENCES ACROSS SYSTEMS
-- ============================================================================

-- SQLite-specific features used in textbook:
-- - AUTOINCREMENT for auto-generating IDs (bookstore_sales, bookstore_inventory)
-- - Dynamic typing (less strict than standard SQL)
-- - substring() function for extracting parts of strings

-- Standard SQL equivalents in other systems:
-- PostgreSQL, Oracle: GENERATED BY DEFAULT AS IDENTITY
-- MySQL: AUTO_INCREMENT
-- SQL Server: IDENTITY property

-- Date handling differences:
-- SQLite: Uses TEXT for dates
-- MySQL: DATE type doesn't support DEFAULT for current date, use TIMESTAMP
-- Others: Standard DATE type with DEFAULT CURRENT_DATE support

-- Reserved keyword issues:
-- MySQL: "condition" is reserved, use backticks `condition` in queries
-- Other systems may have similar issues with column names

-- Character encoding:
-- All data uses UTF-8 encoding
-- Some systems may require explicit encoding settings for correct display
