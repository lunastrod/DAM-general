-- Example table with multiple data types
CREATE TABLE test (
  id INTEGER,
  x  VARCHAR(20),
  y  DATE,
  z  NUMERIC(10,2)
);

-- Insert data into test table
INSERT INTO test VALUES
  (0, 'this is a test', '2021-06-14', 1234.56),
  (1, 'apple', '2021-01-01', 10.10)
;

-- Query test table
SELECT * FROM test;

-- View table definition in SQLite
SELECT sql FROM sqlite_master WHERE name = 'test';

-- DROP TABLE statement
DROP TABLE test;

-- DROP TABLE with IF EXISTS (safer)
DROP TABLE IF EXISTS test;

-- CREATE TABLE AS SELECT (from query results)
DROP TABLE IF EXISTS recent_books;

CREATE TABLE recent_books AS
  SELECT
    a.name AS author,
    b.title,
    b.publication_year
  FROM
    authors AS a
    JOIN books AS b ON a.author_id = b.author_id
  WHERE b.publication_year >= 2010
;

-- View definition of created table
SELECT sql FROM sqlite_master WHERE name = 'recent_books';

-- Query the created table
SELECT * FROM recent_books;

-- CREATE TABLE with DEFAULT values
DROP TABLE IF EXISTS test2;

CREATE TABLE test2 (
  id INTEGER,
  greeting VARCHAR(15) DEFAULT 'Hello'
);

-- Insert with explicit value for default column
INSERT INTO test2 (id, greeting) VALUES (1, 'Good morning');

-- Insert with NULL for default column
INSERT INTO test2 (id, greeting) VALUES (2, NULL);

-- Insert without providing default column (uses default)
INSERT INTO test2 (id) VALUES (3);

SELECT * FROM test2;

-- CREATE TABLE with DEFAULT function (timestamp)
DROP TABLE IF EXISTS test3;

CREATE TABLE test3 (
  purchase VARCHAR(10),
  created_at VARCHAR(20) DEFAULT CURRENT_TIMESTAMP
);

INSERT INTO test3 (purchase) VALUES ('apple');

SELECT * FROM test3;

-- Standard SQL syntax for auto-increment (not in SQLite)
column_name type GENERATED BY DEFAULT AS IDENTITY

-- Alternative standard syntax for auto-increment
column_name type GENERATED ALWAYS AS IDENTITY

-- SQLite-specific AUTOINCREMENT with PRIMARY KEY
DROP TABLE IF EXISTS test4;

CREATE TABLE test4 (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  greeting VARCHAR(15)
);

-- Insert without providing id (auto-generated)
INSERT INTO test4 (greeting) VALUES ('Hello');

-- Insert with explicit id value
INSERT INTO test4 (id, greeting) VALUES (4, 'Good day');

-- Insert without id again (continues from max)
INSERT INTO test4 (greeting) VALUES ('Good afternoon');

SELECT * FROM test4;
