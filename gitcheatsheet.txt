git config --global user.name "Nombre"
git config --global user.email "email@example.com"
git config --global core.editor "code --wait"   # editor por defecto
git config --list                                # ver configuración

git init                        # inicializar repo local
git clone <url>                 # clonar repositorio remoto
git clone <url> <carpeta>       # clonar en carpeta específica

git status                      # estado del working tree
git diff                        # cambios sin stagear
git diff --staged               # cambios en staging
git log                         # historial de commits
git log --oneline --graph       # historial compacto con ramas
git log -p                      # historial con diffs
git show <commit>               # detalles de un commit

git add <archivo>               # añadir archivo al staging
git add .                       # añadir todo
git add -p                      # añadir cambios por partes (interactivo)
git commit -m "mensaje"         # commit con mensaje
git commit -am "mensaje"        # add + commit (solo archivos tracked)
git commit --amend              # modificar el último commit

git branch                      # listar ramas locales
git branch -a                   # listar todas (incluye remotas)
git branch <nombre>             # crear rama
git switch <rama>               # cambiar de rama
git switch -c <rama>            # crear y cambiar de rama
git branch -d <rama>            # borrar rama (seguro)
git branch -D <rama>            # borrar rama (forzado)
git merge <rama>                # fusionar rama en la actual
git rebase <rama>               # rebase sobre otra rama

git remote -v                           # listar remotos
git remote add origin <url>             # añadir remoto
git remote remove <nombre>              # eliminar remoto
git fetch                               # descargar cambios sin integrar
git pull                                # fetch + merge
git pull --rebase                       # fetch + rebase
git push origin <rama>                  # subir rama
git push -u origin <rama>              # subir y establecer upstream
git push --force-with-lease             # push forzado (seguro)

git restore <archivo>           # descartar cambios en working tree
git restore --staged <archivo>  # sacar del staging
git revert <commit>             # revertir commit (crea uno nuevo)
git reset --soft HEAD~1         # deshacer último commit (mantiene staged)
git reset --mixed HEAD~1        # deshacer último commit (mantiene working)
git reset --hard HEAD~1         # deshacer último commit (borra todo)
git clean -fd                   # eliminar archivos no tracked

git stash                       # guardar cambios temporalmente
git stash push -m "mensaje"     # stash con nombre
git stash list                  # ver stashes guardados
git stash pop                   # aplicar último stash y borrarlo
git stash apply stash@{n}       # aplicar stash sin borrarlo
git stash drop stash@{n}        # eliminar stash
git stash clear                 # eliminar todos los stashes

git tag                         # listar tags
git tag <nombre>                # crear tag ligero
git tag -a <nombre> -m "msg"    # crear tag anotado
git push origin <tag>           # subir tag
git push origin --tags          # subir todos los tags
git tag -d <nombre>             # borrar tag local

git merge --abort               # cancelar merge con conflictos
git rebase --abort
git rebase --continue           # continuar tras resolver conflictos
git checkout --ours <archivo>   # quedarse con la versión propia
git checkout --theirs <archivo> # quedarse con la versión entrante
